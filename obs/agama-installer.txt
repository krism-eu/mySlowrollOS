#!/bin/bash
# ==============================================================================
# config.sh - Post-install live system configuration for Agama Installer ISO
# Executed by KIWI inside the chroot environment before packing the image.
# ==============================================================================

test -f /.kconfig && . /.kconfig
test -f /.profile && . /.profile

set -euo pipefail

echo "=== [mySlowrollOS] Configurazione live image Agama Installer ==="

# 0. Impostazione password root per ambiente Live (utente: root / password: root)
echo "root:root" | chpasswd

# 1. Preparazione directory e certificati Agama
mkdir -p -m 0755 /etc/agama.d/ssl /run/agama /etc/pki/trust/anchors

if command -v openssl >/dev/null 2>&1; then
    echo "Generazione certificato SSL locale per Agama / Firefox (localhost)..."
    openssl req -x509 -nodes -days 3650 -newkey rsa:2048 \
        -keyout /etc/agama.d/ssl/server.key \
        -out /etc/agama.d/ssl/server.crt \
        -subj "/CN=localhost" \
        -addext "subjectAltName=DNS:localhost,DNS:agama.local,IP:127.0.0.1" 2>/dev/null || true
    chmod 0600 /etc/agama.d/ssl/server.key 2>/dev/null || true
    chmod 0644 /etc/agama.d/ssl/server.crt 2>/dev/null || true
    cp -f /etc/agama.d/ssl/server.crt /etc/pki/trust/anchors/agama-local.crt 2>/dev/null || true
    if command -v update-ca-certificates >/dev/null 2>&1; then
        update-ca-certificates 2>/dev/null || true
    fi
fi

# 2. Abilitazione servizi di rete, bus di sistema e sessioni logind
systemctl enable NetworkManager.service 2>/dev/null || true
systemctl enable dbus.socket 2>/dev/null || true
systemctl enable systemd-logind.service 2>/dev/null || true

# 3. Abilitazione backend e server web di Agama se presenti
for srv in agama.service agama-web-server.service cockpit.service; do
    if [ -f "/usr/lib/systemd/system/${srv}" ] || [ -f "/etc/systemd/system/${srv}" ]; then
        echo "Abilitazione servizio: ${srv}"
        systemctl enable "${srv}" 2>/dev/null || true
    fi
done

# 3. Configurazione e abilitazione sessione Kiosk grafica locale (Wayland Cage + Firefox)
if [ -f "/usr/lib/systemd/system/agama-kiosk.service" ] || [ -f "/etc/systemd/system/agama-kiosk.service" ]; then
    echo "Abilitazione sessione grafica locale: agama-kiosk.service"
    chmod 0644 /usr/lib/systemd/system/agama-kiosk.service 2>/dev/null || true
    if [ -f "/usr/libexec/myslowroll/wait-for-agama.sh" ]; then
        chmod 0755 /usr/libexec/myslowroll/wait-for-agama.sh
    fi
    systemctl enable agama-kiosk.service 2>/dev/null || true
    systemctl set-default graphical.target 2>/dev/null || true
fi

# 4. Verifica profilo e asset mySlowrollOS (puro JSON per bypassare dipendenza jsonnet)
PROFILE_PATH="/usr/share/myslowroll/profile-final.json"
PROFILE_PATH_LEGACY="/usr/share/myslowroll/profile-final.jsonnet"

if [ ! -f "${PROFILE_PATH}" ]; then
    echo "ERRORE: Profilo mySlowrollOS non trovato in ${PROFILE_PATH}" >&2
    echo "Assicurarsi che root/usr/share/myslowroll/profile-final.json sia presente nell'albero della recipe." >&2
    exit 1
fi

chmod 0644 "${PROFILE_PATH}"
chown root:root "${PROFILE_PATH}"

# Crea symlink di compatibilità .jsonnet -> .json
ln -sf "profile-final.json" "${PROFILE_PATH_LEGACY}"

# Controllo anti-dipendenze esterne: la ISO deve essere 100% autonoma
if grep -q "raw.githubusercontent.com" "${PROFILE_PATH}"; then
    echo "ERRORE CRITICO: Trovati URL remoti verso raw.githubusercontent.com in ${PROFILE_PATH}!" >&2
    echo "La build offline della ISO richiede che tutte le policy siano pacchettizzate in RPM o locali." >&2
    exit 1
fi

echo "=== [mySlowrollOS] Live image configurata con successo (kiosk attivo, zero URL GitHub) ==="
exit 0

# mySlowrollOS — Procedura di Costruzione ISO con KIWI su OBS

Guida completa per creare l'immagine ISO Live Installer di **mySlowrollOS** utilizzando **KIWI NG** e **Agama** su Open Build Service (OBS).

---

## Architettura della ISO Installer

La ISO di installazione combina l'ambiente live ufficiale **Agama** con la personalizzazione completa **mySlowrollOS**:
1. **100% Autonoma (Zero dipendenze esterne durante l'installazione)**:
   - Il profilo di installazione [`profile-final.json`](root/usr/share/myslowroll/profile-final.json) (con symlink di compatibilità `profile-final.jsonnet`) è incorporato nel filesystem live in `/usr/share/myslowroll/profile-final.json`.
   - Tutti i 21 file di configurazione (`myslowroll-policy/`), gli script (`post-chroot-boot-policy.sh`, `init-firstboot.sh`) e le chiavi GPG sono inclusi nella directory overlay `root/usr/share/myslowroll/` e referenziati con URI `file:///usr/share/myslowroll/...`.
2. **Kiosk Grafico Locale Cablato (Wayland Cage + Firefox)**:
   - L'overlay include la systemd unit [`agama-kiosk.service`](root/usr/lib/systemd/system/agama-kiosk.service) (attivata in `graphical.target` su `tty1`), le policy enterprise per i certificati SSL in `/etc/firefox/policies/policies.json` e lo script di sincronizzazione [`wait-for-agama.sh`](root/usr/libexec/myslowroll/wait-for-agama.sh).
   - All'avvio della ISO, Cage avvia automaticamente Firefox in fullscreen kiosk mode su `https://localhost/`, consentendo l'installazione interattiva direttamente sul monitor locale.
   - L'accesso remoto via browser su `https://agama.local` o via porta 443/80 rimane contestualmente attivo su rete locale.
3. **Setup semi-interattivo**:
   - Il kernel cmdline del bootloader avvia Agama con:
     ```text
     inst.auto=file:///usr/share/myslowroll/profile-final.json inst.install=0 inst.systemd_boot_preview=1 live.password=root
     ```
   - `inst.auto=file:///...`: pre-carica automaticamente la lista pacchetti (`criscore1/2`, `myslowroll-cc`, `atomic-update`), kernel, preset, zram e policy di sistema.
   - `inst.install=0`: impedisce l'installazione automatica non presidiata; la configurazione dischi/partizioni e le credenziali utente rimangono **completamente interattive**.
   - `inst.systemd_boot_preview=1`: forza l'uso di systemd-boot/`sdbootutil` come bootloader del sistema installato, coerente con la post-chroot policy del progetto (opzione supportata da Agama v24 in Slowroll).
4. **Repository mySlowrollOS integrato**:
   - I pacchetti protetti e il Control Center sono risolti dal repository `home:krism` (priorità 90).

## Procedura Operativa in 5 Fasi

### Fase 1: Creazione del Pacchetto su OBS

Nel tuo progetto OBS (`home:krism`), crea o apri il pacchetto `agama-installer`:
```bash
osc mkpac home:krism agama-installer
cd home:krism/agama-installer
cp -a /percorso/a/mySlowrollOS/obs/agama-installer/* .
```

---

### Fase 2: Configurazione del Repository OBS (`prjconf`)

Affinché OBS tratti il pacchetto come immagine ISO (e non come un normale RPM), il repository deve avere il tipo configurato su `kiwi`:

1. Nel tuo progetto `home:krism`, modifica la configurazione:
   ```bash
   osc meta prjconf home:krism -e
   ```
2. Aggiungi la direttiva:
   ```text
   Type: kiwi
   ```
3. L'immagine finale generata verrà pubblicata nella sezione **`home:krism/images/`** di download.opensuse.org.

---

### Fase 3: Prerequisiti Pacchetti su OBS

Prima di compilare la ISO, assicurati che i pacchetti `home:krism` siano stati caricati e compilati con successo (`succeeded`):
1. **`myslowroll-cc`**: Sorgenti e spec da `KSCC/dist/`.
2. **`criscore1`**: Spec e sorgenti da `OS/mySlowrollOS/obs/criscore1/`.
3. **`atomic-update`**: Spec e script da `OS/mySlowrollOS/atomic-update/`.

---

### Fase 4: Commit su OBS e Monitoraggio Build

Invia i file al Build Service:
```bash
cd home:krism/agama-installer
osc status
osc addremove
osc commit -m "build(iso): mySlowrollOS Agama Live Installer with offline profile, cage kiosk and branding"
```

Monitora l'avanzamento della build ISO:
```bash
osc results home:krism agama-installer
osc buildlog home:krism agama-installer images x86_64
```

Una volta terminata la compilazione, scarica la ISO generata:
`https://download.opensuse.org/repositories/home:/krism/images/`

---

### Fase 5: Validazione in Macchina Virtuale (Pre-Hardware)

Prima di effettuare la reinstallazione sul disco fisico, esegui il test end-to-end su una VM QEMU/KVM:

```bash
qemu-system-x86_64 \
    -enable-kvm \
    -m 4096 \
    -smp 4 \
    -bios /usr/share/qemu/ovmf-x86_64.bin \
    -drive file=test-vm-disk.qcow2,if=virtio,format=qcow2 \
    -cdrom mySlowrollOS-Agama-installer.x86_64.iso \
    -boot d \
    -vga virtio \
    -device virtio-net-pci,netdev=net0 \
    -netdev user,id=net0
```

#### Checklist di verifica in VM:
- [x] Il bootloader UEFI avvia Agama con `inst.auto=file:///...` pre-caricato.
- [x] Cage avvia Firefox in modalità kiosk visualizzando localmente la schermata di Agama.
- [x] Agama mostra la selezione semi-interattiva: dischi e credenziali configurabili manualmente.
- [x] Nessun errore di download (tutti i file di policy e script sono letti localmente).
- [x] Installazione completata con successo.
- [x] Al primo boot da disco: `systemd-boot` avvia la snapshot RW Btrfs `@/.snapshots/1/snapshot`.
- [x] `myslowroll-cc` si avvia correttamente e visualizza lo stato del sistema e di `atomic-update`.

#!/bin/bash
# ==============================================================================
# wait-for-agama.sh - Attende l'avvio del server HTTP/HTTPS di Agama prima di avviare Firefox
# ==============================================================================
set -euo pipefail

# 0. Garantisce password root impostata a runtime nel sistema live
echo "root:root" | chpasswd 2>/dev/null || true

echo "mySlowrollOS Kiosk: attesa disponibilità del server Agama su localhost..."

for i in {1..60}; do
    if curl -s -f -o /dev/null -m 1 http://localhost/ || curl -s -f -o /dev/null -m 1 http://127.0.0.1/ || curl -s -k -f -o /dev/null -m 1 https://localhost/ || curl -s -k -f -o /dev/null -m 1 https://127.0.0.1/; then
        echo "mySlowrollOS Kiosk: server Agama rilevato su localhost (tentativo $i)."
        echo "root" | agama auth login 2>/dev/null || true
        exit 0
    fi
    sleep 0.5
done

echo "mySlowrollOS Kiosk: avvio Firefox in kiosk mode..."
exit 0

#!/usr/bin/env bash
set -euo pipefail

# First-boot runtime policy for mySlowrollOS.
# The native SDDM/default-target symlinks are already prepared offline by
# post-chroot-boot-policy.sh so the very first boot can reach SDDM directly.

# yast2-bootloader writes /etc/kernel/cmdline AFTER Agama's post/chroot scripts,
# and can inherit failsafe options from the live installer ISO on x86_64.
# Keep only the installed system's ordinary options and enforce AppArmor.
sanitize_kernel_cmdline() (
  # Scope noglob to this subshell: preserve caller options and literal tokens.
  set -f
  local token cleaned=""
  for token in $1; do
    case "$token" in
      ide=nodma|apm=off|noresume|edd=off|nomodeset|3|security=*|systemd.show_status=*|quiet)
        continue ;;
    esac
    cleaned="${cleaned:+$cleaned }$token"
  done
  printf '%s\n' "${cleaned:+$cleaned }security=apparmor systemd.show_status=1 quiet"
)

# Edit only verified stock btrfsmaintenance keys, keeping distro defaults intact.
# The distro's refresh service will own the timer schedule; no custom daemon.
configure_btrfsmaintenance() {
  local config="$1" key
  [[ -f "$config" ]] || return 1
  for key in BTRFS_BALANCE_PERIOD BTRFS_DEFRAG_PERIOD BTRFS_TRIM_PERIOD \
             BTRFS_SCRUB_PERIOD BTRFS_SCRUB_PRIORITY BTRFS_SCRUB_MOUNTPOINTS; do
    [[ "$(grep -Ec "^$key=" "$config")" == 1 ]] || return 1
  done
  sed -i \
    -e 's/^BTRFS_BALANCE_PERIOD=.*/BTRFS_BALANCE_PERIOD="none"/' \
    -e 's/^BTRFS_DEFRAG_PERIOD=.*/BTRFS_DEFRAG_PERIOD="none"/' \
    -e 's/^BTRFS_TRIM_PERIOD=.*/BTRFS_TRIM_PERIOD="none"/' \
    -e 's/^BTRFS_SCRUB_PERIOD=.*/BTRFS_SCRUB_PERIOD="monthly"/' \
    -e 's/^BTRFS_SCRUB_PRIORITY=.*/BTRFS_SCRUB_PRIORITY="idle"/' \
    -e 's@^BTRFS_SCRUB_MOUNTPOINTS=.*@BTRFS_SCRUB_MOUNTPOINTS="/"@' \
    "$config"
}

# Disable startup only, without masking: another dependency may still request
# a unit, and package removal is postponed until the separate RPM audit.
disable_optional_unit() {
  local unit="$1" state
  state="$(systemctl is-enabled "$unit" 2>/dev/null || true)"
  case "$state" in
    enabled|enabled-runtime|linked|linked-runtime)
      if ! systemctl disable "$unit"; then
        echo "myslowroll-firstboot: cannot disable optional unit $unit" >&2
        firstboot_rc=1
      fi
      ;;
    disabled|masked|static|indirect|generated|alias|not-found|'') ;;
    *) echo "myslowroll-firstboot: unknown state for $unit ($state)" >&2; firstboot_rc=1 ;;
  esac
}

# Functional regression tests without privileged actions or running services.
if [[ "${1:-}" == "--self-test" ]]; then
  sample='root=/dev/vda1 rw ide=nodma apm=off noresume edd=off nomodeset 3 mitigations=auto security= rootflags=subvol=0/.snapshots/1/snapshot'
  expected='root=/dev/vda1 rw mitigations=auto rootflags=subvol=0/.snapshots/1/snapshot security=apparmor systemd.show_status=1 quiet'
  [[ "$(sanitize_kernel_cmdline "$sample")" == "$expected" ]] || exit 1
  [[ "$(sanitize_kernel_cmdline "$expected")" == "$expected" ]] || exit 1
  glob_dir="$(mktemp -d)" || exit 1
  touch "$glob_dir/match-one" "$glob_dir/match-two"
  glob_sample="$glob_dir/* $glob_dir/match-? [abc]"
  glob_actual="$(sanitize_kernel_cmdline "$glob_sample")"
  rm -rf -- "$glob_dir"
  [[ "$glob_actual" == "$glob_sample security=apparmor systemd.show_status=1 quiet" ]] || {
    echo 'FAIL: cmdline glob token was expanded' >&2; exit 1;
  }
  # Test the shipped file, not only a hand-picked canonical string.
  shipped_file="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)/myslowroll-policy/kernel-cmdline"
  shipped="$(<"$shipped_file")"
  [[ "$(sanitize_kernel_cmdline "$shipped")" == "$shipped" ]] || {
    echo 'FAIL: shipped cmdline would trigger a needless firstboot rewrite' >&2
    exit 1
  }
  maintenance_test="$(mktemp)" || exit 1
  trap 'rm -f -- "$maintenance_test"' EXIT
  cat >"$maintenance_test" <<'TEST_BTRFS'
BTRFS_BALANCE_PERIOD="weekly"
BTRFS_DEFRAG_PERIOD="monthly"
BTRFS_TRIM_PERIOD="monthly"
BTRFS_SCRUB_PERIOD="weekly"
BTRFS_SCRUB_PRIORITY="normal"
BTRFS_SCRUB_MOUNTPOINTS="auto"
BTRFS_ALLOW_CONCURRENCY="false"
TEST_BTRFS
  configure_btrfsmaintenance "$maintenance_test" || exit 1
  grep -Fxq 'BTRFS_BALANCE_PERIOD="none"' "$maintenance_test" || exit 1
  grep -Fxq 'BTRFS_DEFRAG_PERIOD="none"' "$maintenance_test" || exit 1
  grep -Fxq 'BTRFS_TRIM_PERIOD="none"' "$maintenance_test" || exit 1
  grep -Fxq 'BTRFS_SCRUB_PERIOD="monthly"' "$maintenance_test" || exit 1
  grep -Fxq 'BTRFS_SCRUB_PRIORITY="idle"' "$maintenance_test" || exit 1
  grep -Fxq 'BTRFS_SCRUB_MOUNTPOINTS="/"' "$maintenance_test" || exit 1
  grep -Fxq 'BTRFS_ALLOW_CONCURRENCY="false"' "$maintenance_test" || exit 1
  configure_btrfsmaintenance "$maintenance_test" || exit 1
  [[ "$(grep -c '^BTRFS_BALANCE_PERIOD=' "$maintenance_test")" == 1 ]] || exit 1
  printf '%s\n' 'BTRFS_BALANCE_PERIOD="daily"' >>"$maintenance_test"
  if configure_btrfsmaintenance "$maintenance_test"; then
    echo 'FAIL: duplicate config key accepted' >&2
    exit 1
  fi
  echo 'PASS: Btrfs timer policy, one root scrub, preservation, idempotence'
  echo 'PASS: failsafe removed, root/subvolume preserved, shipped cmdline idempotent'
  # Verify sudoers.d verification logic
  test_sudoers_dir="$(mktemp -d)" || exit 1
  test_sudoers="$test_sudoers_dir/sudoers.d"
  mkdir -m 0750 "$test_sudoers"
  [[ "$(stat -c '%a' "$test_sudoers")" == "750" ]] || { rm -rf -- "$test_sudoers_dir"; exit 1; }
  chmod 0755 "$test_sudoers"
  [[ "$(stat -c '%a' "$test_sudoers")" != "750" ]] || { rm -rf -- "$test_sudoers_dir"; exit 1; }
  rm -rf -- "$test_sudoers_dir"
  echo 'PASS: sudoers.d mode verification logic'
  exit 0
fi

# Repair the installer cmdline before the remaining first-boot policy.
# An sdbootutil failure must be reported but must not skip service setup.
# Storage and EFI choices remain entirely interactive in Agama.
firstboot_rc=0
cmdline_file=/etc/kernel/cmdline
if [[ ! -f "$cmdline_file" ]]; then
  echo "myslowroll-firstboot: missing $cmdline_file" >&2
  firstboot_rc=1
else
  before="$(<"$cmdline_file")"
  after="$(sanitize_kernel_cmdline "$before")"
  if [[ "$before" != "$after" ]]; then
    echo "myslowroll-firstboot: normalizing installed kernel options" >&2
    backup_ok=1
    if [[ ! -e "$cmdline_file.agama-before-repair" ]] &&
       ! cp -a -- "$cmdline_file" "$cmdline_file.agama-before-repair"; then
      echo "myslowroll-firstboot: cannot back up kernel cmdline" >&2
      backup_ok=0
      firstboot_rc=1
    fi
    if (( backup_ok == 1 )); then
      if ! printf '%s\n' "$after" > "$cmdline_file"; then
        echo "myslowroll-firstboot: cannot write kernel cmdline" >&2
        firstboot_rc=1
      elif ! command -v sdbootutil >/dev/null 2>&1; then
        echo "myslowroll-firstboot: sdbootutil missing; boot entries were not updated" >&2
        firstboot_rc=1
      elif ! sdbootutil update-all-entries; then
        echo "myslowroll-firstboot: sdbootutil failed; boot entries may be stale" >&2
        firstboot_rc=1
      fi
    fi
  else
    echo "myslowroll-firstboot: kernel command line already clean"
  fi
  if grep -Eq '(^|[[:space:]])(ide=nodma|apm=off|noresume|edd=off|nomodeset|3|security=)([[:space:]]|$)' "$cmdline_file"; then
    echo "myslowroll-firstboot: failsafe options still present in $cmdline_file" >&2
    firstboot_rc=1
  fi
  if grep -Eq '(^|[[:space:]])(3|nomodeset)([[:space:]]|$)' /proc/cmdline; then
    echo "myslowroll-firstboot: current boot still has failsafe options; restart once for graphical boot" >&2
  fi
fi

if ! systemctl set-default graphical.target; then
  echo 'myslowroll-firstboot: cannot set graphical default target' >&2
  firstboot_rc=1
fi

# Defensive check: keep the native SDDM selector authoritative.
if [[ "$(readlink -f /etc/systemd/system/display-manager.service 2>/dev/null || true)" != "/usr/lib/systemd/system/sddm.service" ]]; then
  systemctl disable --now display-manager-legacy.service >/dev/null 2>&1 || true
  if ! ln -sfn /usr/lib/systemd/system/sddm.service /etc/systemd/system/display-manager.service; then
    echo 'myslowroll-firstboot: cannot select native SDDM' >&2
    firstboot_rc=1
  fi
  if ! systemctl daemon-reload; then
    echo 'myslowroll-firstboot: systemd daemon-reload failed' >&2
    firstboot_rc=1
  fi
fi

# Persist trust for the OBS repository used by criscore/atomic-update.
# Agama's gpgFingerprints authenticates the installation repository but does
# not globally import that key for future package verification.
if [[ -f /etc/zypp/repos.d/home_krism.key ]]; then
  if ! rpm --import /etc/zypp/repos.d/home_krism.key; then
    echo 'myslowroll-firstboot: OBS key import failed' >&2
    firstboot_rc=1
  fi
else
  echo 'myslowroll-firstboot: OBS key missing' >&2
  firstboot_rc=1
fi

# Verify /etc/sudoers.d mode matches the sudo package expectation (0:0:750).
# Agama post-chroot aligns permissions; this is a fail-closed verification only.
if [[ -d /etc/sudoers.d ]]; then
  actual_mode="$(stat -c '%u:%g:%a' /etc/sudoers.d)"
  if [[ "$actual_mode" != "0:0:750" ]]; then
    echo "myslowroll-firstboot: /etc/sudoers.d ownership/mode $actual_mode != 0:0:750" >&2
    firstboot_rc=1
  fi
fi

# Apply intended runtime service policy.
# Enable independently so one missing unit cannot hide the others.
# Do not force-restart AppArmor or replace an active NTP daemon here.
for unit in NetworkManager.service firewalld.service bluetooth.service \
            chronyd.service apparmor.service snapper-cleanup.timer; do
  if ! systemctl enable "$unit"; then
    echo "myslowroll-firstboot: cannot enable $unit" >&2
    firstboot_rc=1
  fi
done
systemctl disable NetworkManager-wait-online.service || true
systemctl disable smartd.service smartd_generate_opts.path || true
systemctl disable snapper-timeline.timer || true
systemctl disable sshd.service sshd.socket || true
systemctl disable display-manager-legacy.service || true
systemctl disable ModemManager.service || true
if ! systemctl mask ModemManager.service; then
  echo "myslowroll-firstboot: cannot mask ModemManager" >&2
  firstboot_rc=1
fi

# Initial radio policy: services stay available and radio switches remain usable.
# Unlike boot timers, this firstboot script runs only once. Changes made in
# Plasma after installation are NOT forcibly reverted at every later boot.
if command -v nmcli >/dev/null 2>&1; then
  if ! systemctl is-active --quiet NetworkManager.service; then
    systemctl start NetworkManager.service || firstboot_rc=1
  fi
  if systemctl is-active --quiet NetworkManager.service; then
    if ! LC_ALL=C nmcli radio wifi off ||
       [[ "$(LC_ALL=C nmcli radio wifi)" != disabled ]]; then
      echo 'myslowroll-firstboot: initial Wi-Fi OFF state not confirmed' >&2
      firstboot_rc=1
    fi
  else
    echo 'myslowroll-firstboot: NetworkManager inactive; Wi-Fi state unverified' >&2
    firstboot_rc=1
  fi
else
  echo 'myslowroll-firstboot: nmcli missing' >&2
  firstboot_rc=1
fi

# BlueZ stays enabled for manual operation. AutoEnable=false covers newly
# detected adapters; start the service now to verify the initial OFF state.
if ! systemctl is-active --quiet bluetooth.service; then
  if ! systemctl start bluetooth.service; then
    echo 'myslowroll-firstboot: cannot start Bluetooth for initial radio check' >&2
    firstboot_rc=1
  fi
fi
if systemctl is-active --quiet bluetooth.service; then
  if ! grep -Eq '^AutoEnable=false$' /etc/bluetooth/main.conf; then
    echo 'myslowroll-firstboot: BlueZ AutoEnable=false missing' >&2
    firstboot_rc=1
  fi
  if command -v bluetoothctl >/dev/null 2>&1; then
    bt_controllers="$(LC_ALL=C bluetoothctl --timeout 5 list 2>/dev/null || true)"
    if grep -q '^Controller ' <<<"$bt_controllers"; then
      if ! LC_ALL=C bluetoothctl --timeout 5 power off >/dev/null 2>&1; then
        echo 'myslowroll-firstboot: Bluetooth controller could not be powered off' >&2
        firstboot_rc=1
      else
        bt_state="$(LC_ALL=C bluetoothctl --timeout 5 show 2>/dev/null || true)"
        if ! grep -Eq '^[[:space:]]*Powered: no([[:space:]]|$)' <<<"$bt_state"; then
          echo 'myslowroll-firstboot: Bluetooth OFF state could not be confirmed' >&2
          firstboot_rc=1
        fi
      fi
    fi
  fi
else
  echo 'myslowroll-firstboot: Bluetooth inactive; initial radio state unverified' >&2
  firstboot_rc=1
fi

# Native Btrfs timer policy approved for this single-disk Btrfs+ext4 PC.
# No periodic balance/defrag/Btrfs TRIM; monthly low-priority root scrub;
# weekly fstrim.timer covers ext4 /home as well.
if configure_btrfsmaintenance /etc/sysconfig/btrfsmaintenance; then
  if ! systemctl start btrfsmaintenance-refresh.service; then
    echo 'myslowroll-firstboot: stock Btrfs timer refresh failed' >&2
    firstboot_rc=1
  fi
  for unit in btrfs-balance.timer btrfs-defrag.timer btrfs-trim.timer; do
    disable_optional_unit "$unit"
  done
  if ! systemctl enable --now btrfs-scrub.timer fstrim.timer; then
    echo 'myslowroll-firstboot: cannot enable scrub/fstrim timers' >&2
    firstboot_rc=1
  fi
else
  echo 'myslowroll-firstboot: unexpected/missing stock Btrfs config' >&2
  firstboot_rc=1
fi

# Only disable optional storage units after confirming the installed layout.
# Agama storage/partitioning remains entirely interactive. On RAID/LVM or
# remote NVMe systems, keep their own boot services. Never mask/remove them.
if ! command -v lsblk >/dev/null 2>&1; then
  echo 'myslowroll-firstboot: lsblk missing, retaining optional storage boot units' >&2
elif ! storage_types="$(lsblk -nr -o TYPE 2>/dev/null)"; then
  echo 'myslowroll-firstboot: cannot inspect disk layout, retaining storage units' >&2
else
  if grep -Eq '^(raid[0-9]*|md)$' <<<"$storage_types"; then
    echo 'myslowroll-firstboot: md RAID detected, retaining md boot units'
  else
    for unit in mdcheck_start.timer mdcheck_continue.timer mdmonitor-oneshot.timer; do
      disable_optional_unit "$unit"
    done
  fi
  if grep -Eq '^(lvm|mpath)$' <<<"$storage_types"; then
    echo 'myslowroll-firstboot: device-mapper volumes detected, retaining LVM units'
  else
    disable_optional_unit lvm2-monitor.service
    disable_optional_unit blk-availability.service
  fi
fi
# NVMe-over-Fabrics detection must fail closed: if the probe is unavailable,
# do not deactivate auto-connections used by a different installation.
if command -v nvme >/dev/null 2>&1; then
  if fabrics="$(LC_ALL=C nvme list-subsys 2>/dev/null)"; then
    if grep -Eq 'trtype=(tcp|rdma|fc)' <<<"$fabrics"; then
      echo 'myslowroll-firstboot: NVMe fabrics detected, retaining autoconnect'
    else
      disable_optional_unit nvmefc-boot-connections.service
      disable_optional_unit nvmf-autoconnect.service
    fi
  fi
fi
# Back In Time is manual-only. XDG Hidden=true override is delivered by Agama;
# keep the Qt application in the menu; never install a backup autostart daemon.
if ! grep -Fxq 'Hidden=true' /etc/xdg/autostart/backintime.desktop; then
  echo 'myslowroll-firstboot: Back In Time autostart suppression missing' >&2
  firstboot_rc=1
fi

# The Agama post-install chroot already applies the tested Snapper policy.
# First boot VERIFY ONLY: no redundant rewrites, no deletion of snapshots.
snapper_root_config=/etc/snapper/configs/root
if [[ -f "$snapper_root_config" ]]; then
  for setting in 'NUMBER_CLEANUP="yes"' 'NUMBER_LIMIT="4-4"' \
                 'NUMBER_LIMIT_IMPORTANT="0-0"' 'NUMBER_MIN_AGE="3600"' \
                 'TIMELINE_CREATE="no"' 'EMPTY_PRE_POST_CLEANUP="yes"'; do
    if ! grep -Fxq "$setting" "$snapper_root_config"; then
      echo "myslowroll-firstboot: Snapper root setting missing: $setting" >&2
      firstboot_rc=1
    fi
  done
else
  echo 'myslowroll-firstboot: Snapper root config missing' >&2
  firstboot_rc=1
fi

exit "$firstboot_rc"

#!/usr/bin/env bash
set -euo pipefail

# Executed in Agama's installed-system chroot AFTER installation.
# Never repartition, format, choose an ESP, or edit firmware BootOrder.

validate_default_entry() {
  local esp="$1" id="$2" entry
  [[ "$id" =~ ^[A-Za-z0-9._+-]+(\.conf)?$ ]] || return 1
  entry="${esp%/}/loader/entries/${id%.conf}.conf"
  [[ -s "$entry" ]] || return 1
  grep -Eq '^linux[[:space:]]+/' "$entry" || return 1
  grep -Eq '^initrd[[:space:]]+/' "$entry" || return 1
  grep -Eq '^options[[:space:]]+.*root=' "$entry" || return 1
  grep -Eq '^options[[:space:]]+.*rootflags=subvol=.*\.snapshots/[0-9]+/snapshot' "$entry"
}

# Agama may mount the EFI System Partition at /boot or /boot/efi.
# Check the exact mountpoint: --target alone can report the parent mount.
find_esp_mountpoint() {
  local candidate info target fstype
  for candidate in /boot/efi /boot; do
    info="$(findmnt -n -o TARGET,FSTYPE --target "$candidate" 2>/dev/null || true)"
    read -r target fstype <<< "$info"
    if [[ "$target" == "$candidate" && "$fstype" == vfat ]]; then
      printf '%s\n' "$candidate"
      return 0
    fi
  done
  return 1
}

# Agama has installed Snapper, but snapperd/systemd does not run in chroot.
# Preserve installer-created SUBVOLUME/QGROUP, changing only verified keys.
configure_snapper_root_file() {
  local config="$1" key
  [[ -f "$config" ]] || { echo "mySlowrollOS: missing Snapper root config: $config" >&2; return 1; }
  for key in NUMBER_CLEANUP NUMBER_LIMIT NUMBER_LIMIT_IMPORTANT NUMBER_MIN_AGE \
             TIMELINE_CREATE EMPTY_PRE_POST_CLEANUP; do
    if [[ "$(grep -Ec "^${key}=" "$config")" != 1 ]]; then
      echo "mySlowrollOS: unexpected Snapper key count: $key" >&2
      return 1
    fi
  done
  sed -i \
    -e 's/^NUMBER_CLEANUP=.*/NUMBER_CLEANUP="yes"/' \
    -e 's/^NUMBER_LIMIT=.*/NUMBER_LIMIT="4-4"/' \
    -e 's/^NUMBER_LIMIT_IMPORTANT=.*/NUMBER_LIMIT_IMPORTANT="0-0"/' \
    -e 's/^NUMBER_MIN_AGE=.*/NUMBER_MIN_AGE="3600"/' \
    -e 's/^TIMELINE_CREATE=.*/TIMELINE_CREATE="no"/' \
    -e 's/^EMPTY_PRE_POST_CLEANUP=.*/EMPTY_PRE_POST_CLEANUP="yes"/' \
    "$config"
}

# Back In Time ships an autostart .desktop that starts pw-cache and must never
# autolaunch on this workstation. The reviewed Hidden=true override is
# delivered by Agama to a neutral path; the package owns
# /etc/xdg/autostart/backintime.desktop, so the override is installed here with
# an atomic replace and verified byte-for-byte before the first reboot.
install_backintime_override() {
  local asset="${BACKINTIME_OVERRIDE_ASSET:-/var/lib/myslowroll/backintime-autostart-disabled.desktop}"
  local dest="${BACKINTIME_OVERRIDE_DEST:-/etc/xdg/autostart/backintime.desktop}"
  local owner="${BACKINTIME_OVERRIDE_OWNER:-0}"
  local group="${BACKINTIME_OVERRIDE_GROUP:-0}"
  local tmp sections
  [[ -f "$asset" && ! -L "$asset" ]] || { echo "mySlowrollOS: backintime override asset missing or not a regular file: $asset" >&2; return 1; }
  sections="$(grep -Ec '^[[:space:]]*\[Desktop Entry\][[:space:]]*$' "$asset")"
  [[ "$sections" -eq 1 ]] || { echo "mySlowrollOS: backintime override must have exactly one [Desktop Entry] section: $asset" >&2; return 1; }
  grep -Fxq 'Hidden=true' "$asset" || { echo "mySlowrollOS: backintime override missing Hidden=true: $asset" >&2; return 1; }
  if grep -Fq 'backintime pw-cache start' "$asset"; then
    echo "mySlowrollOS: backintime override must not contain an autostart pw-cache command: $asset" >&2
    return 1
  fi
  install -d -m 0755 "$(dirname "$dest")"
  tmp="$(mktemp "${dest}.XXXXXX")" || return 1
  install -m 0644 -o "$owner" -g "$group" "$asset" "$tmp" || return 1
  mv -fT "$tmp" "$dest"
  cmp -s "$asset" "$dest" || { echo "mySlowrollOS: backintime override verification failed: $dest" >&2; return 1; }
  echo "mySlowrollOS: Back In Time autostart suppressed (Hidden=true) -> $dest"
}

# Configure AppImage MIME type associations idempotently and nondestructively.
# Merges into /etc/xdg/mimeapps.list preserving any existing system or desktop defaults.
configure_mime_associations() {
  local file="${1:-/etc/xdg/mimeapps.list}"
  install -d -m 0755 "$(dirname "$file")"
  python3 - "$file" <<'PY' || return 1
import sys, configparser

target = sys.argv[1]
cp = configparser.RawConfigParser()
cp.optionxform = str
try:
    cp.read(target, encoding="utf-8")
except Exception:
    pass

for sec in ["Default Applications", "Added Associations"]:
    if not cp.has_section(sec):
        cp.add_section(sec)

cp.set("Default Applications", "application/vnd.appimage", "appimage-executor.desktop")
cp.set("Default Applications", "application/x-iso9660-appimage", "appimage-executor.desktop")
cp.set("Added Associations", "application/vnd.appimage", "appimage-executor.desktop;")
cp.set("Added Associations", "application/x-iso9660-appimage", "appimage-executor.desktop;")

with open(target, "w", encoding="utf-8") as fp:
    cp.write(fp, space_around_delimiters=False)
PY
  chmod 0644 "$file"
  echo "mySlowrollOS: AppImage mime associations merged idempotently -> $file"
}

# Harden and optimize /etc/fstab: umask=0077 on vfat ESP, noatime on Btrfs subvolumes.
# Preserves comments, UUIDs, non-target filesystems, and column layout.
# Performs atomic replacement via tempfile and rename, with fail-closed error propagation.
configure_fstab_mount_options() {
  local fstab="${1:-/etc/fstab}"
  local esp_mount="${2:-}"
  [[ -f "$fstab" && -s "$fstab" ]] || {
    echo "mySlowrollOS: fstab missing or empty: $fstab" >&2
    return 1
  }
  if [[ -z "$esp_mount" ]]; then
    esp_mount="$(find_esp_mountpoint 2>/dev/null || echo /boot/efi)"
  fi

  local tmp
  tmp="$(mktemp "${fstab}.XXXXXX")" || return 1

  python3 - "$fstab" "$tmp" "$esp_mount" <<'PY' || { rm -f "$tmp"; return 1; }
import sys

src, dst, esp_target = sys.argv[1], sys.argv[2], sys.argv[3]
with open(src, "r", encoding="utf-8") as f:
    lines = f.readlines()

if not lines:
    sys.exit("mySlowrollOS: empty fstab source")

out = []
for line in lines:
    stripped = line.strip()
    if not stripped or stripped.startswith("#"):
        out.append(line)
        continue

    parts = line.split()
    if len(parts) < 4:
        out.append(line)
        continue

    spec, mnt, fstype, opts = parts[0], parts[1], parts[2], parts[3]
    tail = parts[4:] if len(parts) > 4 else ["0", "0"]
    if len(tail) == 1:
        tail.append("0")

    opt_list = opts.split(",")

    if fstype == "vfat" and (mnt == esp_target or mnt in ("/boot/efi", "/boot")):
        new_opts = [o for o in opt_list if not o.startswith("umask=")]
        new_opts.append("umask=0077")
        opts = ",".join(new_opts)
    elif fstype == "btrfs":
        new_opts = [o for o in opt_list if o not in ("relatime", "strictatime", "noatime")]
        new_opts.append("noatime")
        opts = ",".join(new_opts)

    col0 = spec.ljust(43)
    col1 = mnt.ljust(12)
    col2 = fstype.ljust(6)
    col3 = opts.ljust(28)
    out.append(f"{col0} {col1} {col2} {col3} {tail[0]}  {tail[1]}\n")

with open(dst, "w", encoding="utf-8") as f:
    f.writelines(out)
    f.flush()
PY

  chmod 0644 "$tmp"
  mv -fT "$tmp" "$fstab" || { rm -f "$tmp"; return 1; }
}

# Realign and verify sudo package permissions and the custom passwordless drop-in.
# Sudo package ships /etc/sudoers.d with 0750 (root:root).
configure_sudoers_policy() {
  local sudoers_dir="${1:-/etc/sudoers.d}"
  local dropin="${2:-${sudoers_dir}/99-myslowroll-passwordless-admin}"
  local owner="${3:-0}"
  local group="${4:-0}"

  [[ -d "$sudoers_dir" && ! -L "$sudoers_dir" ]] || {
    echo "mySlowrollOS: sudoers directory missing or is a symlink: $sudoers_dir" >&2
    return 1
  }

  if command -v rpm >/dev/null 2>&1 && rpm -q sudo >/dev/null 2>&1; then
    rpm --setperms sudo >/dev/null 2>&1 || true
    rpm --setugids sudo >/dev/null 2>&1 || true
  fi

  chown "$owner:$group" "$sudoers_dir" || return 1
  chmod 0750 "$sudoers_dir" || return 1

  if [[ -f "$dropin" ]]; then
    [[ ! -L "$dropin" ]] || {
      echo "mySlowrollOS: sudoers drop-in is a symlink: $dropin" >&2
      return 1
    }
    chown "$owner:$group" "$dropin" || return 1
    chmod 0440 "$dropin" || return 1
    if command -v visudo >/dev/null 2>&1; then
      visudo -cf "$dropin" >/dev/null 2>&1 || {
        echo "mySlowrollOS: visudo syntax check failed for: $dropin" >&2
        return 1
      }
    fi
  fi
  echo "mySlowrollOS: /etc/sudoers.d permissions (0750) and drop-in verified"
}

#!/usr/bin/env python3
"""Synchronize assets and generate offline profile for Agama KIWI installer.

Copies all policy files, scripts, and keys into root/usr/share/myslowroll/
and transforms profile-final.jsonnet to reference local file:/// URLs.
Supports --check mode for CI without modifying files on disk.
"""
import argparse
import json
import os
import pathlib
import shutil
import stat
import sys
import tarfile


def compute_offline_profile(profile_path):
    offline_profile = json.loads(profile_path.read_text())

    for entry in offline_profile.get("files", []):
        url = entry.get("url", "")
        if "raw.githubusercontent.com" in url:
            parts = url.split("raw.githubusercontent.com/krism-eu/mySlowrollOS/")[1].split("/", 1)
            rel_path = parts[1]
            entry["url"] = f"file:///usr/share/myslowroll/{rel_path}"

    for kind, scripts in offline_profile.get("scripts", {}).items():
        for script in scripts:
            url = script.get("url", "")
            if "raw.githubusercontent.com" in url:
                parts = url.split("raw.githubusercontent.com/krism-eu/mySlowrollOS/")[1].split("/", 1)
                rel_path = parts[1]
                script["url"] = f"file:///usr/share/myslowroll/{rel_path}"

    return json.dumps(offline_profile, indent=2) + "\n"


def pack_root_archive(script_dir):
    root_dir = script_dir / "root"
    archive_path = script_dir / "root.tar.gz"

    with tarfile.open(archive_path, "w:gz") as tar:
        for dirpath, dirs, files in os.walk(root_dir):
            dirs.sort()
            files.sort()
            for d in dirs:
                full_path = os.path.join(dirpath, d)
                rel_path = os.path.relpath(full_path, root_dir)
                info = tar.gettarinfo(full_path, arcname=rel_path)
                info.uid = 0
                info.gid = 0
                info.uname = "root"
                info.gname = "root"
                info.mode = 0o755
                tar.addfile(info)
            for f in files:
                full_path = os.path.join(dirpath, f)
                rel_path = os.path.relpath(full_path, root_dir)
                info = tar.gettarinfo(full_path, arcname=rel_path)
                info.uid = 0
                info.gid = 0
                info.uname = "root"
                info.gname = "root"
                if os.path.islink(full_path):
                    info.type = tarfile.SYMTYPE
                    info.linkname = os.readlink(full_path)
                    tar.addfile(info)
                else:
                    if os.access(full_path, os.X_OK):
                        info.mode = 0o755
                    else:
                        info.mode = 0o644
                    with open(full_path, "rb") as fp:
                        tar.addfile(info, fp)
    return archive_path


def verify_root_archive(archive_path):
    if not archive_path.is_file():
        return ["root.tar.gz mancante"]
    errors = []
    try:
        with tarfile.open(archive_path, "r:gz") as tar:
            names = tar.getnames()
            bad = [n for n in names if n.startswith("root/") or n.startswith("/") or n.startswith("./")]
            if bad:
                errors.append(f"root.tar.gz contiene prefissi non validi ({bad[:3]}...)")
            required = [
                "usr/share/myslowroll/profile-final.json",
                "usr/lib/systemd/system/agama-kiosk.service",
                "usr/libexec/myslowroll/wait-for-agama.sh",
                "etc/firefox/policies/policies.json",
                "usr/lib/firefox/distribution/policies.json",
            ]
            for req in required:
                if req not in names:
                    errors.append(f"root.tar.gz non contiene il file essenziale: {req}")
    except Exception as e:
        errors.append(f"Impossibile leggere root.tar.gz: {e}")
    return errors


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--check", action="store_true", help="fail if overlay assets or profile drift from repository")
    args = parser.parse_args()

    script_dir = pathlib.Path(__file__).resolve().parent
    repo_root = script_dir.parents[1]
    overlay_dir = script_dir / "root" / "usr" / "share" / "myslowroll"

    policy_path = repo_root / "agama" / "profile-policy.json"
    profile_path = repo_root / "agama" / "profile-final.jsonnet"
    if not policy_path.is_file() or not profile_path.is_file():
        print(f"ERROR: Missing {policy_path} or {profile_path}", file=sys.stderr)
        return 1

    policy = json.loads(policy_path.read_text())

    assets = [entry["source"] for entry in policy["files"]]
    for scripts in policy["scripts"].values():
        assets.extend(entry["source"] for entry in scripts)

    expected_profile_content = compute_offline_profile(profile_path)

    if args.check:
        stale = []
        if not overlay_dir.is_dir():
            print(f"ERROR: Overlay directory missing: {overlay_dir}", file=sys.stderr)
            return 1

        for rel_path in assets:
            src = repo_root / rel_path
            dst = overlay_dir / rel_path
            if not dst.is_file():
                stale.append(f"missing asset: {rel_path}")
                continue
            if src.read_bytes() != dst.read_bytes():
                stale.append(f"content mismatch: {rel_path}")
                continue
            expected_mode = 0o755 if src.suffix == ".sh" else 0o644
            actual_mode = stat.S_IMODE(dst.stat().st_mode)
            if actual_mode != expected_mode:
                stale.append(f"permission mismatch: {rel_path} ({oct(actual_mode)} != {oct(expected_mode)})")

        dst_profile = overlay_dir / "profile-final.json"
        if not dst_profile.is_file():
            stale.append("missing profile: profile-final.json")
        elif dst_profile.read_text() != expected_profile_content:
            stale.append("content mismatch: profile-final.json")

        # Verify Kiosk systemd unit and launcher script
        kiosk_service = script_dir / "root" / "usr" / "lib" / "systemd" / "system" / "agama-kiosk.service"
        if not kiosk_service.is_file():
            stale.append("missing kiosk unit: root/usr/lib/systemd/system/agama-kiosk.service")
        else:
            kiosk_text = kiosk_service.read_text()
            if "WLR_LIBINPUT_NO_DEVICES" in kiosk_text:
                stale.append("kiosk unit contains obsolete WLR_LIBINPUT_NO_DEVICES directive (breaks mouse/keyboard!)")
            if "--private-window" in kiosk_text:
                stale.append("kiosk unit contains --private-window (breaks Firefox token/certs persistence!)")
            if "https://localhost/" not in kiosk_text:
                stale.append("kiosk unit does not use https://localhost/")

        kiosk_wait = script_dir / "root" / "usr" / "libexec" / "myslowroll" / "wait-for-agama.sh"
        if not kiosk_wait.is_file():
            stale.append("missing kiosk launcher: root/usr/libexec/myslowroll/wait-for-agama.sh")
        elif not os.access(kiosk_wait, os.X_OK):
            stale.append("kiosk launcher not executable: root/usr/libexec/myslowroll/wait-for-agama.sh")

        # Verify Firefox Enterprise Policies
        ff_policy = script_dir / "root" / "etc" / "firefox" / "policies" / "policies.json"
        if not ff_policy.is_file():
            stale.append("missing firefox policy: root/etc/firefox/policies/policies.json")

        # Verify root.tar.gz archive
        archive_path = script_dir / "root.tar.gz"
        archive_errors = verify_root_archive(archive_path)
        stale.extend(archive_errors)

        if stale:
            print("ERROR: Out-of-date or drifted overlay assets in root/usr/share/myslowroll/:", file=sys.stderr)
            for err in stale:
                print(f"  - {err}", file=sys.stderr)
            return 1

        print(f"PASS: {len(assets)} assets and offline profile match repository sources")
        print(f"PASS: root.tar.gz verificato con successo (nessun prefisso non valido, tutti i file essenziali inclusi)")
        return 0

    # Write / sync mode
    if overlay_dir.exists():
        shutil.rmtree(overlay_dir)
    overlay_dir.mkdir(parents=True, exist_ok=True)

    copied = 0
    for rel_path in assets:
        src = repo_root / rel_path
        dst = overlay_dir / rel_path
        if not src.is_file():
            print(f"ERROR: Source asset not found: {src}", file=sys.stderr)
            return 1
        dst.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src, dst)
        if src.suffix == ".sh":
            dst.chmod(0o755)
        else:
            dst.chmod(0o644)
        copied += 1

    dst_profile = overlay_dir / "profile-final.json"
    dst_profile.write_text(expected_profile_content)
    dst_profile.chmod(0o644)

    dst_profile_legacy = overlay_dir / "profile-final.jsonnet"
    if dst_profile_legacy.is_symlink() or dst_profile_legacy.is_file():
        dst_profile_legacy.unlink()
    dst_profile_legacy.symlink_to("profile-final.json")

    # Self-validation
    if "raw.githubusercontent.com" in expected_profile_content:
        print("ERROR: Remote GitHub URLs still present in offline profile!", file=sys.stderr)
        return 1

    offline_data = json.loads(expected_profile_content)
    for entry in offline_data.get("files", []):
        url = entry.get("url", "")
        if url.startswith("file:///usr/share/myslowroll/"):
            local_rel = url[len("file:///usr/share/myslowroll/"):]
            if not (overlay_dir / local_rel).is_file():
                print(f"ERROR: Referenced file missing in overlay: {local_rel}", file=sys.stderr)
                return 1

    for kind, scripts in offline_data.get("scripts", {}).items():
        for script in scripts:
            url = script.get("url", "")
            if url.startswith("file:///usr/share/myslowroll/"):
                local_rel = url[len("file:///usr/share/myslowroll/"):]
                if not (overlay_dir / local_rel).is_file():
                    print(f"ERROR: Referenced script missing in overlay: {local_rel}", file=sys.stderr)
                    return 1

    print(f"PASS: {copied} assets copied to root/usr/share/myslowroll/")
    print("PASS: Offline profile generated at root/usr/share/myslowroll/profile-final.json (zero GitHub URLs, 100% local references validated)")

    # Pacchettizzazione root.tar.gz per OBS (con path relativi 'usr/...' senza prefisso 'root/')
    arch_file = pack_root_archive(script_dir)
    arch_errors = verify_root_archive(arch_file)
    if arch_errors:
        print(f"ERROR: Verifica archivio {arch_file} fallita:", file=sys.stderr)
        for err in arch_errors:
            print(f"  - {err}", file=sys.stderr)
        return 1
    print(f"PASS: {arch_file.name} impacchettato e verificato con successo per OBS")
    return 0


if __name__ == "__main__":
    sys.exit(main())
